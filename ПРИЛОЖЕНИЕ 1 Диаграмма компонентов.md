```mermaid
flowchart TB

    App["Мобильное приложение"]

    API["API Gateway"]

    subgraph Services["Основные сервисы АС"]

        Order["Заказы"]

    end

    DB[(База данных)]
    Cache[(Кэш)]

    Payment["Платёжный сервис"]
    Delivery["Система доставки"]
    Push["Push-уведомления"]

    App --> API

    API --> Auth
    API --> Catalog
    API --> Cart
    API --> Order
    API --> Loyalty
    API --> Support
    API --> Notify

    Auth --> DB
    Catalog --> DB
    Cart --> DB
    Order --> DB
    Loyalty --> DB
    Support --> DB

    Catalog --> Cache
    Cart --> Cache

    Order --> Payment
    Order --> Delivery
    ```
